local llt, lle = nil, false
local GE = RS:FindFirstChild("GrabEvents")
local bhC = 0
local bhS = {}

local function isBH(o)
    local n = o.Name:lower()
    return n:find("blackhole") or n:find("black_hole") or n:find("black hole")
end
local function hasBA(o)
    local p = o.Parent
    while p and p ~= WS do
        if isBH(p) then return true end
        p = p.Parent
    end
    return false
end
local function ckBH(o)
    if not isBH(o) or hasBA(o) or bhS[o] then return end
    bhS[o] = true
    bhC = bhC + 1
    BH.Text = "BH: " .. bhC
end
for _, o in ipairs(WS:GetDescendants()) do ckBH(o) end
WS.DescendantAdded:Connect(ckBH)

local function startLag()
    if lle then return end
    lle = true
    llt = task.spawn(function()
        if not GE then return end
        local cl = GE:FindFirstChild("CreateGrabLine")
        if not cl then return end
        while lle do
            local t = WS:FindFirstChild("SpawnLocation") or WS:FindFirstChild("Spawn") or (LP.Character and LP.Character:FindFirstChild("HumanoidRootPart"))
            if t then
                for i = 1, 10 do
                    pcall(function()
                        cl:FireServer(t, CFrame.new(math.random(-1e9, 1e9), 0, math.random(-1e9, 1e9)))
                    end)
                end
            end
            task.wait(0.05)
        end
    end)
end
local function stopLag()
    lle = false
    if llt then task.cancel(llt) end
end

local cb, ia = nil, false
local function allP()
    local t = {}
    for _, p in ipairs(P:GetPlayers()) do
        if p ~= LP then table.insert(t, p) end
    end
    return t
end
local function mr()
    return LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
end
local function fix(tr, pos)
    if not tr then return end
    for _, v in ipairs(tr:GetChildren()) do
        if v:IsA("BodyPosition") or v:IsA("BodyGyro") then v:Destroy() end
    end
    local bp = Instance.new("BodyPosition")
    bp.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    bp.P = _G.FixP
    bp.D = _G.FixD
    bp.Position = pos
    bp.Parent = tr
    local bg = Instance.new("BodyGyro")
    bg.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    bg.P = _G.FixP
    bg.D = _G.FixD
    bg.CFrame = CFrame.new(pos)
    bg.Parent = tr
end
local function unfix()
    for _, p in ipairs(P:GetPlayers()) do
        if p ~= LP then
            local h = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if h then
                for _, v in ipairs(h:GetChildren()) do
                    if v:IsA("BodyPosition") or v:IsA("BodyGyro") then v:Destroy() end
                end
            end
        end
    end
end
local function spawnBlob()
    local r = mr()
    if not r then return nil end
    local mt = RS:FindFirstChild("MenuToys")
    if not mt then return nil end
    local st = mt:FindFirstChild("SpawnToyRemoteFunction")
    if not st then return nil end
    pcall(function()
        st:InvokeServer("CreatureBlobman", r.CFrame * CFrame.new(0, 0, -5), Vector3.new(0, 127, 0))
    end)
    task.wait(0.5)
    local tf = WS:FindFirstChild(LP.Name .. "SpawnedInToys")
    return tf and tf:FindFirstChild("CreatureBlobman")
end
local function sit(b)
    if not b or not LP.Character then return end
    local v = b:FindFirstChild("VehicleSeat")
    if not v then return end
    local h = LP.Character:FindFirstChildOfClass("Humanoid")
    if h then pcall(function() v:Sit(h) end) end
    task.wait(0.3)
end

local function kickSingle(t)
    if ia or not t or t == LP then return end
    ia = true
    cb = spawnBlob()
    if not cb then ia = false return end
    sit(cb)
    local r = mr()
    if not r then ia = false return end
    local sc = cb:FindFirstChild("BlobmanSeatAndOwnerScript")
    if not sc then ia = false return end
    local t0 = os.clock()
    while os.clock() - t0 < 8 and t.Parent ~= nil do
        local tr = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
        if tr then
            pcall(function()
                r.CFrame = tr.CFrame
                r.AssemblyLinearVelocity = Vector3.zero
            end)
            task.wait()
            pcall(function()
                sc.CreatureGrab:FireServer(cb.LeftDetector, tr, cb.LeftDetector.LeftWeld)
                sc.CreatureGrab:FireServer(cb.RightDetector, tr, cb.RightDetector.RightWeld)
            end)
            task.wait(0.03)
            local lw = cb.LeftDetector:FindFirstChild("LeftWeld")
            local rw = cb.RightDetector:FindFirstChild("RightWeld")
            if lw and lw.Part1 == tr and rw and rw.Part1 == tr then break end
        else
            task.wait(0.05)
        end
    end
    local r2 = mr()
    local tr2 = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
    if r2 and tr2 then fix(tr2, r2.CFrame * CFrame.new(0, 2, 0).Position) end
    while t.Parent ~= nil do
        local tr = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
        local r3 = mr()
        if tr and r3 then
            local p = r3.CFrame * CFrame.new(0, 2, 0).Position
            pcall(function()
                tr.CFrame = CFrame.new(p)
                tr.AssemblyLinearVelocity = Vector3.zero
                tr.AssemblyAngularVelocity = Vector3.zero
            end)
            pcall(function()
                sc.CreatureGrab:FireServer(cb.LeftDetector, tr, cb.LeftDetector.LeftWeld)
                sc.CreatureGrab:FireServer(cb.RightDetector, tr, cb.RightDetector.RightWeld)
            end)
        end
        task.wait(0.03)
    end
    if cb and cb.Parent then pcall(function() cb:Destroy() end) end
    local tf = WS:FindFirstChild(LP.Name .. "SpawnedInToys")
    if tf then
        local b = tf:FindFirstChild("CreatureBlobman")
        if b then pcall(function() b:Destroy() end) end
    end
    cb = nil
    ia = false
end

local function kickAll()
    if ia then return end
    ia = true
    local ap = allP()
    if #ap == 0 then ia = false return end
    cb = spawnBlob()
    if not cb then ia = false return end
    sit(cb)
    local r = mr()
    if not r then ia = false return end
    for _, p in ipairs(ap) do
        local tr = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
        if tr then
            r.CFrame = tr.CFrame
            r.AssemblyAngularVelocity = Vector3.zero
            task.wait()
            if (r.Position - tr.Position).Magnitude <= 35 then
                for i = 1, 8 do
                    if not tr.Parent or not cb.Parent then break end
                    pcall(function()
                        cb.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(cb.LeftDetector, tr, cb.LeftDetector.LeftWeld)
                        cb.BlobmanSeatAndOwnerScript.CreatureRelease:FireServer(cb.LeftDetector.LeftWeld)
                    end)
                    task.wait()
                end
            end
        end
    end
    r.CFrame = CFrame.new(0, 100, 0)
    r.AssemblyLinearVelocity = Vector3.zero
    task.wait(0.1)
    for _, v in ipairs(cb:GetDescendants()) do
        if v:IsA("BasePart") then pcall(function() v.Anchored = true end) end
    end
    task.wait(0.1)
    local rad, h = 25, 110
    for i, p in ipairs(ap) do
        local tr = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
        if tr then
            local a = math.rad((i - 1) * (360 / #ap))
            local pos = Vector3.new(rad * math.cos(a), h, rad * math.sin(a))
            tr.CFrame = CFrame.new(pos)
            fix(tr, pos)
        end
    end
    task.wait(0.1)
    for _ = 1, 2 do
        for _, p in ipairs(ap) do
            local tr = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if tr then
                task.spawn(function()
                    pcall(function()
                        RS.GrabEvents.SetNetworkOwner:FireServer(tr, CFrame.new(tr.Position))
                        RS.GrabEvents.DestroyGrabLine:FireServer(tr)
                    end)
                end)
            end
        end
        task.wait(0.1)
    end
    task.wait(0.3)
    for _, p in ipairs(ap) do
        local tr = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
        if tr then
            task.spawn(function()
                pcall(function()
                    cb.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(cb.LeftDetector, tr, cb.LeftDetector.LeftWeld)
                    cb.BlobmanSeatAndOwnerScript.CreatureGrab:FireServer(cb.RightDetector, tr, cb.RightDetector.RightWeld)
                end)
            end)
        end
    end
    task.wait(0.1)
    for _, v in ipairs(cb:GetDescendants()) do
        if v:IsA("BasePart") then pcall(function() v.Anchored = false end) end
    end
    task.wait(1)
    if cb and cb.Parent then pcall(function() cb:Destroy() end) end
    local tf = WS:FindFirstChild(LP.Name .. "SpawnedInToys")
    if tf then
        local b = tf:FindFirstChild("CreatureBlobman")
        if b then pcall(function() b:Destroy() end) end
    end
    cb = nil
    ia = false
end

local function startSpamKick(targetPlr)
    if not targetPlr then return end
    spamKickOn = true
    spamKickTarget = targetPlr
    startLagSpam()
    
    local GE = RS:FindFirstChild("GrabEvents")
    local SetNO = GE and GE:FindFirstChild("SetNetworkOwner")
    local DestroyGL = GE and GE:FindFirstChild("DestroyGrabLine")
    
    spamKickConn = task.spawn(function()
        while spamKickOn and spamKickTarget and spamKickTarget.Parent do
            local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
            local tRoot = spamKickTarget.Character and spamKickTarget.Character:FindFirstChild("HumanoidRootPart")
            if myRoot and tRoot then
                if (myRoot.Position - tRoot.Position).Magnitude > 25 then
                    pcall(function() myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3) end)
                end
                pcall(function()

        end

                        if h then

local function drag(ob)
    local d, s, sp = false, nil, nil
    ob.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
            d = true
            s = i.Position
            sp = ob.Position
        end
    end)
    ob.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
            d = false
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if d and (i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseMovement) then
            local dl = i.Position - s
            ob.Position = UDim2.new(sp.X.Scale, sp.X.Offset + dl.X, sp.Y.Scale, sp.Y.Offset + dl.Y)
        end
    end)
end
